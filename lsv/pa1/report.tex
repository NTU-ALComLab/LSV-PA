\section{Exercise 2: Using ABC}

\subsection*{2(a) Two-bit multiplier BLIF}
File: \texttt{mul.blif}. Inputs $A=(a_1 a_0)$, $B=(b_1 b_0)$; outputs $Y=(y_3 y_2 y_1 y_0)=A\cdot B$.
Partial products $p_0{=}a_0b_0$, $p_1{=}a_0b_1$, $p_2{=}a_1b_0$, $p_3{=}a_1b_1$, then
$y_0{=}p_0$, $y_1{=}p_1\oplus p_2$, $y_2{=}p_3\oplus(p_1p_2)$, $y_3{=}p_3(p_1p_2)$.

\section{Exercise 3: Boolean Function Representations}

\subsection*{3(a) Comparisons}

\paragraph{AIG (\texttt{aig}) vs structurally hashed AIG (\texttt{strash}).}
\texttt{aig} rewrites each node's local function into an AIG while keeping the original network topology ($nd{=}9$, reported $aig{=}12$, $lev{=}3$).
\texttt{strash} builds a single structurally hashed AIG; duplicate AND nodes are shared ($and{=}12$, $lev{=}4$).
So \texttt{strash} is the global hashed DAG; \texttt{aig} is local AIG functions on the old netlist.

\hfill
\includegraphics[width=0.45\textwidth]{screenshots/ex3_strash.png}
\caption{Left: after \texttt{aig}. Right: after \texttt{strash}.}

\paragraph{BDD (\texttt{bdd}) vs collapsed BDD (\texttt{collapse}).}
\texttt{bdd} attaches a local BDD to each logic node ($nd{=}9$, $bdd{=}16$, $lev{=}3$).
\texttt{collapse} builds global BDDs for the POs and replaces the network by those functions ($nd{=}4$, $bdd{=}14$, $lev{=}1$).
Collapsed BDDs remove intermediate nodes and expose the full I/O function.

\subsection*{3(b) Strashed AIG $\rightarrow$ SOP logic network}
Command sequence used:
\begin{lstlisting}
read mul.blif
strash
logic
sop
print_stats
print_factor
\end{lstlisting}
\texttt{logic} converts the AIG back to a logic network; \texttt{sop} expresses each node in sum-of-products.
Result: $nd{=}12$, $cube{=}12$, $lev{=}4$, with factored forms printable by \texttt{print\_factor}.
